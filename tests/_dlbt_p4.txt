def test_dual_lb_dedup_independent(tmp_path, monkeypatch):
    """(b) Dedup keys include lb — lb48 open does not suppress lb72."""
    sys.path.insert(0, str(ROOT / "scripts"))
    import zeus_wedge_1h_shadow as sh

    monkeypatch.setattr(sh, "SHADOW_PATH", tmp_path / "signals.jsonl")
    monkeypatch.setattr(sh, "STATE_PATH", tmp_path / "state.json")

    assert sh._dedup_key("BTC-USDT", "narrow", "long", 48) != sh._dedup_key(
        "BTC-USDT", "narrow", "long", 72
    )
    assert sh._dedup_key("BTC-USDT", "narrow", "long", 48) == "BTC-USDT|narrow|long|48"

    # Seed done for lb=48 only
    st = sh._load_state()
    sec48 = sh._lb_section(st, 48)
    sec48["done"]["BTC-USDT|narrow|long|48"] = int(__import__("time").time() * 1000)
    sh._save_state(st)

    o, h, l, c, _form = _narrow_series(lb=48)
    base_ts = 1_700_000_000
    pad = [
        {
            "open": 100.0,
            "high": 101.0,
            "low": 99.0,
            "close": 100.0,
            "volume": 500.0,
            "open_time": base_ts + i * 3600,
        }
        for i in range(80)
    ]
    bars = pad + [
        {
            "open": o[i],
            "high": h[i],
            "low": l[i],
            "close": c[i],
            "volume": 1000.0,
            "open_time": base_ts + (80 + i) * 3600,
        }
        for i in range(len(o))
    ]
    last = bars[-1]
    bars.append(
        {
            "open": last["close"],
            "high": last["close"],
            "low": last["close"],
            "close": last["close"],
            "volume": 1,
            "open_time": last["open_time"] + 3600,
        }
    )
    sh.log_wedge_1h_shadow(symbol="BTC-USDT", bars_1h=bars, lbs=(48, 72))
    lines = [json.loads(ln) for ln in sh.SHADOW_PATH.read_text().splitlines() if ln.strip()]
    # lb=48 narrow long signal suppressed by done; other statuses/geoms/lb72 still ok
    sig48_narrow_long = [
        r
        for r in lines
        if r.get("lb") == 48
        and r.get("geom") == "narrow"
        and r.get("side") == "long"
        and r.get("status") == "signal"
    ]
    assert not sig48_narrow_long, sig48_narrow_long
    assert any(r.get("lb") == 72 for r in lines), lines
