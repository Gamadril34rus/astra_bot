def test_secrets_not_in_redact():
    # mirror live_env._redact without importing aiohttp chain
    def _redact(s: object) -> str:
        t = str(s or "")
        if len(t) > 8:
            return t[:3] + "***" + t[-2:]
        return "***"
    assert "***" in _redact("abcdefghijklmnop")
    assert "abcdefghijklmnop" not in _redact("abcdefghijklmnop")
